    <script>
        const rates = {
            newLF: 0.45,
            restripeLF: 0.28,
            adaStencils: 45.00,
            arrowStencils: 25.00,
            curbLF: 1.75,
            blackoutLF: 0.65,
            baseMobilization: 350.00,
            mileageRate: 2.20,
            freeDistance: 30.0,
            afterHoursMultiplier: 1.25,
            prevailingWageMultiplier: 1.35
        };

        function calculateEstimate() {
            const val = id => Number(document.getElementById(id).value) || 0;
            const checked = id => document.getElementById(id).checked;

            const subtotal = 
                (val('newLF') * rates.newLF) +
                (val('restripeLF') * rates.restripeLF) +
                (val('adaStencils') * rates.adaStencils) +
                (val('arrowStencils') * rates.arrowStencils) +
                (val('curbLF') * rates.curbLF) +
                (val('blackoutLF') * rates.blackoutLF) +
                (Math.max(0, val('travelMiles') - rates.freeDistance) * rates.mileageRate);

            let total = Math.max(subtotal, rates.baseMobilization);

            if (checked('afterHours')) total *= rates.afterHoursMultiplier;
            if (checked('prevailingWage')) total *= rates.prevailingWageMultiplier;

            document.getElementById('totalValue').innerText = '$' + total.toLocaleString('en-US', {
                minimumFractionDigits: 2,
                maximumFractionDigits: 2
            });
        }

        const fieldIds = ['newLF', 'restripeLF', 'adaStencils', 'arrowStencils', 'curbLF', 'blackoutLF', 'travelMiles', 'afterHours', 'prevailingWage'];
        fieldIds.forEach(id => {
            const el = document.getElementById(id);
            if (el) {
                el.addEventListener('input', calculateEstimate);
                el.addEventListener('change', calculateEstimate);
            }
        });

        calculateEstimate();
    </script>
